import subprocess
import json
import os
from pathlib import Path

CLI_PATH = os.getenv("CLI_PATH", "../jz-claude-skills/skills/diagnosis-orchestrator/scripts/diagnosis_cli.py")

def run_analysis(case_dir: str) -> dict:
    """调 CLI 执行 analyze 子命令"""
    result = subprocess.run(
        ["python", CLI_PATH, "analyze", case_dir, "--format", "json"],
        capture_output=True, text=True, timeout=120
    )
    if result.returncode != 0:
        raise CLIError(result.stderr)
    return json.loads(result.stdout)

def run_collect(source_type: str, params: dict, case_dir: str) -> dict:
    """调 CLI 执行数据采集"""
    cmd = ["python", CLI_PATH, "run", "--source", source_type, "--case-dir", case_dir]
    if params.get("robot_ip"):
        cmd += ["--ip", params["robot_ip"]]
    if params.get("task_url"):
        cmd += ["--task-url", params["task_url"]]
    if params.get("symptom"):
        cmd += ["--symptom", params["symptom"]]
    
    result = subprocess.run(cmd, capture_output=True, text=True, timeout=300)
    if result.returncode != 0:
        raise CLIError(result.stderr)
    return json.loads(result.stdout)

class CLIError(Exception):
    pass

# knowledge/playbook_loader.py
import yaml
from pathlib import Path

PLAYBOOK_DIR = Path("../jz-claude-skills/../nav-manager/docs/jstate_error_codes/nav-navigation-errors")

def load_playbook(code: str) -> dict | None:
    """加载单个错误码的 Playbook"""
    path = PLAYBOOK_DIR / f"{code}-*.md"
    matches = list(PLAYBOOK_DIR.glob(f"{code}-*.md"))
    if not matches:
        return None
    
    content = matches[0].read_text()
    # 提取 YAML playbook 段
    yaml_start = content.find("```yaml\nplaybook:")
    if yaml_start == -1:
        return None
    yaml_block = content[yaml_start + 7:content.find("```", yaml_start + 7)]
    return yaml.safe_load(yaml_block)

def load_all_playbooks() -> dict[str, dict]:
    """加载全部 Playbook → {code: playbook}"""
    playbooks = {}
    for path in PLAYBOOK_DIR.glob("*.md"):
        code = path.name.split("-")[0]
        playbook = load_playbook(code)
        if playbook:
            playbooks[code] = playbook
    return playbooks
```

### Step 4.2：加载错误码关联图

```python
# knowledge/correlation.py
import yaml

GRAPH_PATH = Path(__file__).parent / "correlation_graph.yaml"

def load_correlation_graph() -> dict:
    return yaml.safe_load(GRAPH_PATH.read_text()) if GRAPH_PATH.exists() else {}

def resolve_priorities(found_codes: list[str]) -> list[dict]:
    """拓扑排序，上游优先"""
    graph = load_correlation_graph()
    edges = {e["upstream"]: e["downstream"] for e in graph.get("edges", [])}
    
    queue = []
    for code in found_codes:
        if code in edges and edges[code] in found_codes:
            queue.append({"code": code, "priority": "LOW", 
                         "reason": f"可能是 {edges[code]} 的下游症状"})
        else:
            queue.append({"code": code, "priority": "HIGH", "reason": "上游或无依赖"})
    
    # 共现检查
    patterns = graph.get("co_occurrence_patterns", [])
    for pattern in patterns:
        if all(c in found_codes for c in pattern["codes"]):
            queue.insert(0, {"code": "CO_OCCURRENCE", "priority": "HIGHEST",
                            "reason": pattern["hint"]})
    
    return sorted(queue, key=lambda x: {"HIGHEST": 0, "HIGH": 1, "LOW": 2}[x["priority"]])
```

### Step 5.2：`field_agent`（多线程）

```python
# agent/nodes/field_agent.py
import asyncio
from agent.state import DiagnosisState
from models.router import route

TOP_K_TYPES = ["navigation_rotate", "safety", "task_chain", "device_sensor", "navigation"]

async def field_agent(state: DiagnosisState) -> DiagnosisState:
    """现场 Agent：TopK 多线程并行探索 + 历史统计"""
    symptom = state.get("symptom", state.get("user_input", ""))
    site = state.get("extra_params", {}).get("site", "unknown")
    
    # 多线程并行
    tasks = [_explore_thread(symptom, ptype, site) for ptype in TOP_K_TYPES[:3]]
    thread_results = await asyncio.gather(*tasks)
    
    # 汇总：加权排序
    all_findings = []
    for result in thread_results:
        if result:
            all_findings.append(result)
    
    all_findings.sort(key=lambda x: x.get("score", 0), reverse=True)
    
    state["field_agent_result"] = {
        "threads": thread_results,
        "top_finding": all_findings[0] if all_findings else None,
        "root_cause": all_findings[0]["conclusion"] if all_findings else "unknown",
        "confidence": "high" if all_findings and all_findings[0].get("score", 0) > 5 else "medium",
    }
    return state

async def _explore_thread(symptom: str, problem_type: str, site: str) -> dict | None:
    """单个探索线程"""

现象: {symptom}
站点: {site}

搜索历史 Case 中该类型的匹配记录。输出 JSON:
{{"problem_type": "", "history_count": 0, "top_cause": "", "top_cause_ratio": 0.0,

# engine/file_manager.py
import asyncio
from pathlib import Path

class FileAccessManager:
    def __init__(self, max_concurrent_io: int = 2):
        self._cache: dict[str, bytes] = {}
        self._locks: dict[str, asyncio.Lock] = {}
        self._io_semaphore = asyncio.Semaphore(max_concurrent_io)
    
    async def read(self, path: str, offset: int = 0, size: int = -1) -> bytes:
        if path in self._cache:
            data = self._cache[path]
            return data[offset:offset+size] if size > 0 else data[offset:]
        
        async with self._io_semaphore:
            lock = self._locks.setdefault(path, asyncio.Lock())
            async with lock:
                if path not in self._cache:
                    self._cache[path] = await asyncio.to_thread(Path(path).read_bytes)
        
        return self._cache[path][offset:offset+size]
    
    async def replay_bag(self, path: str, topic: str, duration: int) -> dict:
        async with self._io_semaphore:
            # TODO: rosbag play 实现
            return {"topic": topic, "messages": 0}
    
    def invalidate(self, path: str):
        self._cache.pop(path, None)
```

### Session Manager

# server/session_manager.py
from datetime import datetime, timezone
import uuid

class SessionManager:
    def __init__(self):
        self._cases: dict[str, dict] = {}
    
    def create(self, user_id: str, params: dict) -> str:
        case_id = f"{user_id}-{uuid.uuid4().hex[:8]}"
        self._cases[case_id] = {
            "case_id": case_id,
            "user_id": user_id,
            "status": "pending",
            "created_at": datetime.now(timezone.utc).isoformat(),
            "params": params,
            "progress": [],
        }
        return case_id
    
    def get(self, case_id: str, user_id: str) -> dict | None:
        case = self._cases.get(case_id)
        if case and case["user_id"] == user_id:
            return case
        return None
    
    def list_by_user(self, user_id: str) -> list[dict]:
        return [c for c in self._cases.values() if c["user_id"] == user_id]
```

## 11. Phase 10：诊断闭环（1 天）

### 目标

CLOSED Case 自动入库，统计触发 Playbook 增强建议。

### Step 10.1：Case 入库

```python
# engine/precipitation.py
import json
from pathlib import Path

CASE_INDEX = Path("data/stats/case_index.jsonl")

def index_case(case_id: str, conclusion: dict):
    """CLOSED Case → 写入检索池"""
    entry = {
        "case_id": case_id,
        "problem_type": conclusion.get("problem_type", ""),
        "root_cause": conclusion.get("root_cause", ""),
        "symptom": conclusion.get("symptom", ""),
        "conclusion_status": conclusion.get("status", ""),
        "closed_at": conclusion.get("closed_at", ""),
    }
    with open(CASE_INDEX, "a") as f:
        f.write(json.dumps(entry) + "\n")
```

### Step 10.2：统计检查

```python
def check_playbook_enhancement():
    """同根因 ≥ 5 次 → 建议增强"""
    from collections import Counter
    
    causes = []
    with open(CASE_INDEX) as f:
        for line in f:
            entry = json.loads(line)
            causes.append(entry["root_cause"])
    
    for cause, count in Counter(causes).items():
        if count >= 5:
            print(f"[建议] 根因 '{cause}' 出现 {count} 次，建议增强对应 Playbook")
```

### 验收

- [ ] Case CLOSED 后自动写入 `case_index.jsonl`
- [ ] 手动跑 `check_playbook_enhancement()` 输出增强建议

## 13. 常见问题

**Q: CLI 路径找不到？**
确认 `.env` 中 `CLI_PATH` 指向 `jz-claude-skills/skills/diagnosis-orchestrator/scripts/diagnosis_cli.py`。

**Q: DeepSeek API 返回 401？**
检查 `.env` 中的 `DEEPSEEK_API_KEY`。用 `curl` 直接验证。

**Q: LangGraph 节点没执行？**
检查 `graph.py` 中条件边的返回值是否匹配 `add_conditional_edges` 中注册的 key。

**Q: SSE 流中断？**
确认 Nginx（如果有）配置了 `proxy_buffering off` 和 `chunked_transfer_encoding on`。
